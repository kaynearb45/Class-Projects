// This pseudocode displays employee net pay values.
// All employees have a standard $45 deduction.
// If an employee does not earn enough to cover the deduction,
// a message is displayed.

start
   Declarations
      string name
      string EOFNAME = "ZZZ"
      num hours
      num rate
      num DEDUCTION = 45
      num gross
      num net

   output "Enter first name or ", EOFNAME, " to quit"
   input name

   while name <> EOFNAME
      output "Enter hours worked for ", name
      input hours

      output "Enter hourly rate for ", name
      input rate

      gross = hours * rate
      net = gross - DEDUCTION

      if net > 0 then
         output "Net pay for ", name, " is ", net
      else
         output "Deductions not covered. Net is 0."
      endif

      output "Enter next name or ", EOFNAME, " to quit"
      input name
   endwhile

   output "End of job"
stop
